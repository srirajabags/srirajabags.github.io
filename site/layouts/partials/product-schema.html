{{/* One catalog product as a schema.org Product, for its own page. Takes (dict "page" "p" "url" ) and returns the Product dict; the caller adds @context and prints it. Only call it for a PRICED product: Google's product snippets require offers, and a fabricated price is never the fix. */}} {{- $page := .page }}{{ $p := .p }}{{ $url := .url -}} {{ $prod := dict "@type" "Product" "name" $p.name "url" $url "brand" (dict "@type" "Brand" "name" $page.Site.Title) "manufacturer" (dict "@type" "Organization" "@id" (printf "%s#business" $page.Site.BaseURL)) }} {{ with $p.note }}{{ $prod = merge $prod (dict "description" .) }}{{ end }} {{/* All photos, not just the card one — Google accepts an array and prefers several angles of the same product. */}} {{ with $p.imgs }} {{ $urls := slice }} {{ range . }}{{ $urls = $urls | append (.full | absURL) }}{{ end }} {{ $prod = merge $prod (dict "image" $urls) }} {{ end }} {{ with $p.shops }}{{ $prod = merge $prod (dict "category" (delimit . ", ")) }}{{ end }} {{/* One row can be several variants. The GSMs and the extra sizes are published as properties rather than as separate Products: they share a name, an image and a page anchor, so splitting them would put near-duplicate entries in the list for no gain. */}} {{ $props := slice }} {{/* Material comes from the row. It used to be hardcoded as non-woven polypropylene, which was wrong the moment plastic and bio-degradable covers joined the sheet. */}} {{ with $p.materials }} {{ $props = $props | append (dict "@type" "PropertyValue" "name" "Material" "value" (delimit . ", ")) }} {{ end }} {{ with $p.gsms }} {{ $g := slice }}{{ range . }}{{ $g = $g | append (string .value) }}{{ end }} {{ $props = $props | append (dict "@type" "PropertyValue" "name" (cond (eq $p.thickness "micron") "Thickness" "GSM") "value" (delimit $g ", ") "unitText" $p.thickness) }} {{ end }} {{ with $p.patties }} {{ $props = $props | append (dict "@type" "PropertyValue" "name" "Side patty" "value" (delimit . ", ")) }} {{ end }} {{ if gt (len $p.sizes) 1 }} {{ $s := slice }}{{ range $p.sizes }}{{ $s = $s | append .plain }}{{ end }} {{ $props = $props | append (dict "@type" "PropertyValue" "name" "Available sizes" "value" (delimit $s ", ")) }} {{ end }} {{ with $p.colours }} {{ $cl := slice }}{{ range . }}{{ $cl = $cl | append .name }}{{ end }} {{ $props = $props | append (dict "@type" "PropertyValue" "name" "Colour" "value" (delimit $cl ", ")) }} {{ end }} {{ with $props }}{{ $prod = merge $prod (dict "additionalProperty" .) }}{{ end }} {{/* width/height describe a single artefact, so they are only emitted when the row really is one size. */}} {{ if eq (len $p.sizes) 1 }} {{ $s := index $p.sizes 0 }} {{ if $s.ok }} {{ $prod = merge $prod (dict "width" (dict "@type" "QuantitativeValue" "value" $s.w "unitCode" "INH") "height" (dict "@type" "QuantitativeValue" "value" $s.h "unitCode" "INH")) }} {{ end }} {{ end }} {{/* Rates vary by size, GSM, side patty and order value. A product with one rate gets an Offer with that price; one with a spread gets an AggregateOffer from its lowest rate to its highest, which is what Google reads for a price range. The unit — per kg or per piece — rides along as a UnitPriceSpecification, and the minimum order as eligibleQuantity. */}} {{ $unitCode := cond (eq $p.unit "pc") "H87" "KGM" }} {{ $spec := dict "@type" "UnitPriceSpecification" "priceCurrency" "INR" "referenceQuantity" (dict "@type" "QuantitativeValue" "value" 1 "unitCode" $unitCode) }} {{ $offer := dict "priceCurrency" "INR" "availability" "https://schema.org/InStock" "url" $url "seller" (dict "@type" "Organization" "@id" (printf "%s#business" $page.Site.BaseURL)) }} {{ if eq $p.rateMin $p.rateMax }} {{ $offer = merge $offer (dict "@type" "Offer" "price" $p.rateMin "priceSpecification" (merge $spec (dict "price" $p.rateMin))) }} {{ else }} {{ $n := 0 }}{{ range $p.variants }}{{ if .has }}{{ $n = add $n 1 }}{{ end }}{{ end }} {{ $offer = merge $offer (dict "@type" "AggregateOffer" "lowPrice" $p.rateMin "highPrice" $p.rateMax "offerCount" $n "priceSpecification" (merge $spec (dict "minPrice" $p.rateMin "maxPrice" $p.rateMax))) }} {{ end }} {{ if $p.hasMoq }} {{ $offer = merge $offer (dict "eligibleQuantity" (dict "@type" "QuantitativeValue" "minValue" $p.moq "unitCode" $unitCode)) }} {{ end }} {{ $prod = merge $prod (dict "offers" $offer) }} {{- return $prod -}}
