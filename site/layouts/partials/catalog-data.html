{{/* Catalog data layer. Reads assets/data/catalog.csv — the one file the office edits. It is plain CSV precisely so it can be opened in Google Sheets, filled in, exported and dropped back here unchanged. Nothing else in the build hard-codes a product. Three rules follow from that: 1. Columns are matched by HEADER NAME, not position. Reordering columns in the sheet, or adding a column we do not read yet, cannot break the page. 2. LIST COLUMNS hold several values separated by "|" — sizes, gsms, colours, shop_types, badges, images. One row is one product, however many variants it comes in, so the sheet never needs a row per size. The first image is the one the card shows; the rest become the carousel on the product's detail view. 3. ALIGNED COLUMNS carry one value per entry of the list they belong to: pcs_per_kg aligns to sizes ("44|58" against "12X16|10X14") rate_per_kg aligns to gsms ("165|180" against "60|70") A single value applies to every entry, which is the common case. A short list fills what it covers and leaves the rest unpriced rather than recycling values — a wrong rate is worse than a missing one. Every commercial number stays optional. A blank cell is the normal state, not an error: the page falls back to "Rate on request" and hides the arithmetic that depends on it. That is what lets the catalog go live before the rate card is settled. Facet lists are derived from the rows rather than declared, so a new GSM typed into the sheet shows up as a filter on the next build. Returns: dict of products, colours, facets, shopTiles, hasAnyRate. */}} {{- $stepKg := 25 -}} {{/* A cell -> clean list of values. Empty entries are dropped, so trailing pipes and "60||70" from a sloppy paste are harmless. */}} {{- define "partials/catalog-list" -}} {{- $out := slice -}} {{- range (split . "|") -}} {{- $v := trim . " " -}} {{- if $v }}{{ $out = $out | append $v }}{{ end -}} {{- end -}} {{- return $out -}} {{- end -}} {{/* Same, but for ALIGNED columns, where position carries meaning. Empty slots are KEPT: against gsms "60|70", a rate of "165|" prices the 60 and leaves the 70 on request. Dropping the blank would collapse this to a single value and silently price both — the kind of error nobody spots in a spreadsheet. */}} {{- define "partials/catalog-slots" -}} {{- $out := slice -}} {{- if trim . " " -}} {{- range (split . "|") }}{{ $out = $out | append (trim . " ") }}{{ end -}} {{- end -}} {{- return $out -}} {{- end -}} {{/* Number, read leniently: a sheet round-trip can add currency symbols, thousands separators or stray spaces. Strip to digits and a decimal point, then accept only if what remains is a real number. Anything else — "n/a", "TBC", "0" — is "not supplied" rather than zero. */}} {{- define "partials/catalog-num" -}} {{- $raw := replaceRE "[^0-9.]" "" (string .) -}} {{- $n := 0.0 -}} {{- if findRE "^[0-9]+(\\.[0-9]+)?$" $raw }}{{ $n = float $raw }}{{ end -}} {{- return $n -}} {{- end -}} {{/* ---- colour lookup: name -> swatch hex ------------------------------- */}} {{/* Keyed lowercase so "navy" and "Navy" in the product sheet both resolve. A blank or non-hex value is not an error: the swatch falls back to a CSS class keyed on the colour's slug, which is how Multicolour gets its gradient without putting CSS inside a data file. */}} {{- $colours := dict -}} {{- with resources.Get "data/catalog-colours.csv" -}} {{- $rows := . | transform.Unmarshal -}} {{- range after 1 $rows -}} {{- if ge (len .) 1 -}} {{- $name := trim (index . 0) " ﻿" -}} {{- $hex := "" -}} {{- if ge (len .) 2 }}{{ $hex = trim (index . 1) " " }}{{ end -}} {{/* Only a literal hex colour is allowed through to a style attribute. */}} {{- if not (findRE "^#[0-9a-fA-F]{3,8}$" $hex) }}{{ $hex = "" }}{{ end -}} {{- if $name }}{{ $colours = merge $colours (dict (lower $name) $hex) }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{/* ---- products -------------------------------------------------------- */}} {{- $products := slice -}} {{- $hasAnyRate := false -}} {{- $hasAnyMoq := false -}} {{- with resources.Get "data/catalog.csv" -}} {{- $rows := . | transform.Unmarshal -}} {{- if gt (len $rows) 1 -}} {{/* Header row, normalised. ﻿ strips the BOM Google Sheets prepends on export, which would otherwise make the first column name unmatchable. */}} {{- $head := slice -}} {{- range index $rows 0 -}} {{- $head = $head | append (lower (trim . " ﻿")) -}} {{- end -}} {{- $order := 0 -}} {{- range after 1 $rows -}} {{- $r := . -}} {{/* Row -> map. Short rows are fine; missing cells read as absent. */}} {{- $m := dict -}} {{- range $i, $k := $head -}} {{- if and $k (lt $i (len $r)) -}} {{- $m = merge $m (dict $k (trim (index $r $i) " ")) -}} {{- end -}} {{- end -}} {{- $name := index $m "name" | default "" -}} {{- if $name -}} {{- $order = add $order 1 -}} {{- $id := index $m "id" | default "" -}} {{- if not $id }}{{ $id = urlize $name }}{{ end -}} {{- $badges := partial "catalog-list" (index $m "badges" | default "") -}} {{- $shops := partial "catalog-list" (index $m "shop_types" | default "") -}} {{/* What the bag is actually made of — non woven, plastic, bio-degradable. Read as a list like the rest, so a row could carry more than one. */}} {{- $materials := partial "catalog-list" (index $m "material" | default "") -}} {{- $moq := 0 }}{{ $hasMoq := false -}} {{- $mq := partial "catalog-num" (index $m "min_order_kg" | default "") -}} {{- if gt $mq 0.0 }}{{ $moq = int $mq }}{{ $hasMoq = true }}{{ $hasAnyMoq = true }}{{ end -}} {{/* ---- sizes, with pcs/kg aligned to them ---- */}} {{- $sizeRaw := partial "catalog-list" (index $m "sizes" | default "") -}} {{- $pcsRaw := partial "catalog-slots" (index $m "pcs_per_kg" | default "") -}} {{- $sizes := slice -}} {{- $areaMin := 0.0 }}{{ $areaMax := 0.0 -}} {{- $buckets := slice -}} {{- range $i, $s := $sizeRaw -}} {{/* "12X16", "12x16", "12 x 16" and "12×16" all mean the same thing. Anything trailing the two numbers is a qualifier the office wrote deliberately — "16x19 (Stick Bag)" — so it is carried through to the label rather than parsed away. */}} {{- $norm := replaceRE "[×✕✖*]" "x" (lower $s) -}} {{- $dims := replaceRE "^\\s*([0-9.]+)\\s*x\\s*([0-9.]+).*$" "$1|$2" $norm -}} {{- $w := 0.0 }}{{ $h := 0.0 }}{{ $suffix := "" -}} {{- if ne $dims $norm -}} {{- $parts := split $dims "|" -}} {{- $w = partial "catalog-num" (index $parts 0) -}} {{- $h = partial "catalog-num" (index $parts 1) -}} {{/* Taken off the ORIGINAL string so the qualifier keeps its case. */}} {{- $suffix = trim (replaceRE "(?i)^\\s*[0-9.]+\\s*[x×✕✖*]\\s*[0-9.]+\\s*" "" $s) " " -}} {{- end -}} {{- $ok := and (gt $w 0.0) (gt $h 0.0) -}} {{- $label := $s }}{{ $plain := $s -}} {{- if $ok -}} {{- $label = printf "%v″ × %v″" $w $h -}} {{- $plain = printf "%v x %v inch" $w $h -}} {{- with $suffix -}} {{- $label = printf "%s %s" $label . -}} {{- $plain = printf "%s %s" $plain . -}} {{- end -}} {{- end -}} {{/* Bucketed on height — the dimension a shopkeeper judges a bag by. A product offered in several sizes belongs to several buckets. */}} {{- $bucket := "" -}} {{- if $ok -}} {{- if le $h 12.0 }}{{ $bucket = "Small" -}} {{- else if le $h 16.0 }}{{ $bucket = "Medium" -}} {{- else }}{{ $bucket = "Large" }}{{ end -}} {{- if not (in $buckets $bucket) }}{{ $buckets = $buckets | append $bucket }}{{ end -}} {{- $a := mul $w $h -}} {{- if or (eq $areaMin 0.0) (lt $a $areaMin) }}{{ $areaMin = $a }}{{ end -}} {{- if gt $a $areaMax }}{{ $areaMax = $a }}{{ end -}} {{- end -}} {{/* pcs/kg for THIS size: one value covers all, otherwise index-aligned. */}} {{- $pcs := 0 -}} {{- if eq (len $pcsRaw) 1 -}} {{- $pcs = int (partial "catalog-num" (index $pcsRaw 0)) -}} {{- else if lt $i (len $pcsRaw) -}} {{- $pcs = int (partial "catalog-num" (index $pcsRaw $i)) -}} {{- end -}} {{- $sizes = $sizes | append (dict "label" $label "plain" $plain "w" $w "h" $h "ok" $ok "area" (mul $w $h) "bucket" $bucket "pcs" $pcs "hasPcs" (gt $pcs 0)) -}} {{- end -}} {{/* ---- gsms, with rate/kg aligned to them ---- */}} {{- $gsmRaw := partial "catalog-list" (index $m "gsms" | default "") -}} {{- $rateRaw := partial "catalog-slots" (index $m "rate_per_kg" | default "") -}} {{- $gsms := slice -}} {{- $rateMin := 0.0 }}{{ $rateMax := 0.0 }}{{ $hasRate := false -}} {{- range $i, $g := $gsmRaw -}} {{- $gv := int (partial "catalog-num" $g) -}} {{- if gt $gv 0 -}} {{- $rate := 0.0 -}} {{- if eq (len $rateRaw) 1 -}} {{- $rate = partial "catalog-num" (index $rateRaw 0) -}} {{- else if lt $i (len $rateRaw) -}} {{- $rate = partial "catalog-num" (index $rateRaw $i) -}} {{- end -}} {{- if gt $rate 0.0 -}} {{- $hasRate = true }}{{ $hasAnyRate = true -}} {{- if or (eq $rateMin 0.0) (lt $rate $rateMin) }}{{ $rateMin = $rate }}{{ end -}} {{- if gt $rate $rateMax }}{{ $rateMax = $rate }}{{ end -}} {{- end -}} {{- $gsms = $gsms | append (dict "value" $gv "rate" $rate "hasRate" (gt $rate 0.0)) -}} {{- end -}} {{- end -}} {{/* ---- colours ---- */}} {{- $cols := slice -}} {{- range partial "catalog-list" (index $m "colours" | default "") -}} {{- $cols = $cols | append (dict "name" . "slug" (urlize .) "hex" (index $colours (lower .) | default "")) -}} {{- end -}} {{/* Images. A pipe-separated list, first entry is the one the card shows and the rest fill the detail carousel. A missing or misspelled filename is skipped rather than failing the build, so one bad name costs that photo and not the page; a row whose names are all bad falls back to a placeholder tile. "image" is still read as a fallback so a sheet exported before the column was renamed keeps working. */}} {{- $imgCell := index $m "images" | default "" -}} {{- if not $imgCell }}{{ $imgCell = index $m "image" | default "" }}{{ end -}} {{- $imgs := slice -}} {{- range partial "catalog-list" $imgCell -}} {{- with resources.Get (printf "images/%s" .) -}} {{- $imgs = $imgs | append (dict "src" (.Fill "320x320 Smart webp q80").RelPermalink "src2x" (.Fill "640x640 Smart webp q80").RelPermalink "full" (.Fill "900x900 Smart webp q82").RelPermalink) -}} {{- end -}} {{- end -}} {{/* The card and anything else wanting "the" photo use the first. */}} {{- $img := "" }}{{ $img2x := "" -}} {{- if $imgs -}} {{- $img = (index $imgs 0).src }}{{ $img2x = (index $imgs 0).src2x -}} {{- end -}} {{- $products = $products | append (dict "id" $id "order" $order "name" $name "img" $img "img2x" $img2x "imgs" $imgs "bagType" (index $m "bag_type" | default "") "print" (index $m "print" | default "") "sizes" $sizes "buckets" $buckets "areaMin" $areaMin "areaMax" $areaMax "gsms" $gsms "colours" $cols "shops" $shops "materials" $materials "badges" $badges "rateMin" $rateMin "rateMax" $rateMax "hasRate" $hasRate "moq" $moq "hasMoq" $hasMoq "startKg" (cond $hasMoq $moq $stepKg) "note" (index $m "note" | default "") ) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{/* ---- facets, derived from the rows above ----------------------------- */}} {{- $types := slice }}{{ $prints := slice }}{{ $gsmVals := slice }}{{ $shopVals := slice }}{{ $colNames := slice }}{{ $matVals := slice }}{{ $badgeVals := slice -}} {{- range $products -}} {{- range .materials }}{{ if not (in $matVals .) }}{{ $matVals = $matVals | append . }}{{ end }}{{ end -}} {{- range .badges }}{{ if not (in $badgeVals .) }}{{ $badgeVals = $badgeVals | append . }}{{ end }}{{ end -}} {{- with .bagType }}{{ if not (in $types .) }}{{ $types = $types | append . }}{{ end }}{{ end -}} {{- with .print }}{{ if not (in $prints .) }}{{ $prints = $prints | append . }}{{ end }}{{ end -}} {{- range .gsms }}{{ if not (in $gsmVals .value) }}{{ $gsmVals = $gsmVals | append .value }}{{ end }}{{ end -}} {{- range .shops }}{{ if not (in $shopVals .) }}{{ $shopVals = $shopVals | append . }}{{ end }}{{ end -}} {{- range .colours }}{{ if not (in $colNames .name) }}{{ $colNames = $colNames | append .name }}{{ end }}{{ end -}} {{- end -}} {{- $gsmVals = sort $gsmVals -}} {{- $shopVals = sort $shopVals -}} {{/* Size options are a fixed ladder, but only the buckets actually stocked are offered — a filter that can only ever return nothing is worse than none. */}} {{- $sizeOpts := slice -}} {{- range (slice (dict "value" "Small" "label" "Small (up to 12″)") (dict "value" "Medium" "label" "Medium (13–16″)") (dict "value" "Large" "label" "Large (17″ and up)")) -}} {{- $v := .value }}{{ $used := false -}} {{- range $products }}{{ if in .buckets $v }}{{ $used = true }}{{ end }}{{ end -}} {{- if $used }}{{ $sizeOpts = $sizeOpts | append . }}{{ end -}} {{- end -}} {{- $facets := slice -}} {{/* Highlights lead the panel. They cut across every other facet — a Bestseller can be any material or shape — so they work as a quick way in before the buyer narrows by what the bag actually is. Sorted, so the list does not reshuffle when a row is edited. */}} {{- if $badgeVals -}} {{- $o := slice }}{{ range (sort $badgeVals) }}{{ $o = $o | append (dict "value" . "label" .) }}{{ end -}} {{- $facets = $facets | append (dict "key" "badge" "title" "Highlights" "options" $o) -}} {{- end -}} {{/* Then material: whether a buyer wants a non woven bag or a plastic cover is the widest cut through the catalog, ahead of shape or print. */}} {{- if $matVals -}} {{- $o := slice }}{{ range $matVals }}{{ $o = $o | append (dict "value" . "label" .) }}{{ end -}} {{- $facets = $facets | append (dict "key" "material" "title" "Material" "options" $o) -}} {{- end -}} {{- if $types -}} {{- $o := slice }}{{ range $types }}{{ $o = $o | append (dict "value" . "label" .) }}{{ end -}} {{- $facets = $facets | append (dict "key" "type" "title" "Bag / cover type" "options" $o) -}} {{- end -}} {{- if $prints -}} {{- $o := slice }}{{ range $prints }}{{ $o = $o | append (dict "value" . "label" .) }}{{ end -}} {{- $facets = $facets | append (dict "key" "print" "title" "Printed or plain" "options" $o) -}} {{- end -}} {{- if $sizeOpts -}} {{- $facets = $facets | append (dict "key" "size" "title" "Size" "options" $sizeOpts) -}} {{- end -}} {{- if $gsmVals -}} {{- $o := slice }}{{ range $gsmVals }}{{ $o = $o | append (dict "value" (string .) "label" (printf "%d GSM" .)) }}{{ end -}} {{- $facets = $facets | append (dict "key" "gsm" "title" "GSM" "options" $o) -}} {{- end -}} {{/* Colour options carry their swatch, so the filter can show the fabric rather than just name it — "Turquoise" and "Pink" mean little to a buyer choosing by eye. */}} {{- if $colNames -}} {{- $o := slice -}} {{- range $colNames -}} {{- $o = $o | append (dict "value" . "label" . "slug" (urlize .) "hex" (index $colours (lower .) | default "")) -}} {{- end -}} {{- $facets = $facets | append (dict "key" "colour" "title" "Colour" "options" $o) -}} {{- end -}} {{- if $shopVals -}} {{- $o := slice }}{{ range $shopVals }}{{ $o = $o | append (dict "value" . "label" .) }}{{ end -}} {{- $facets = $facets | append (dict "key" "shop" "title" "Shop type" "options" $o) -}} {{- end -}} {{/* Shop-type tiles. The representative image is the first product carrying that shop type, so the strip needs no curation and cannot reference an image the catalog does not have. A product counts under each of its shop types. */}} {{- $shopTiles := slice -}} {{- range $shopVals -}} {{- $s := . }}{{ $img := "" }}{{ $n := 0 -}} {{- range $products -}} {{- if in .shops $s -}} {{- $n = add $n 1 -}} {{- if not $img }}{{ $img = .img }}{{ end -}} {{- end -}} {{- end -}} {{- $shopTiles = $shopTiles | append (dict "name" $s "img" $img "count" $n) -}} {{- end -}} {{- return (dict "products" $products "colours" $colours "facets" $facets "shopTiles" $shopTiles "stepKg" $stepKg "hasAnyRate" $hasAnyRate "hasAnyMoq" $hasAnyMoq) -}}
