{{/* Catalog data layer. Reads assets/data/catalog.json. (mul $w $h) "bucket" $bucket) -}} {{- end -}} {{/* Rupees as the price list writes them: whole rupees bare, part rupees to two places — ₹255, ₹13.50. */}} {{- define "partials/catalog-money" -}} {{- -}} {{- return $out -}} {{- end -}} {{- $data := dict -}} {{- with resources.Get "data/catalog.json" }}{{ $data = . | transform.Unmarshal }}{{ end -}} {{/* ---- price types: Type 1, 2, 3 "hex" (index $colours (lower .) | default "")) -}} {{- end -}} {{/* ---- photos ---- A photo carries the shop types it was shot for: the same D-cut bag printed for a textile shop, a medical shop, a sweet stall. "bagType" ($m.bagType | default "") "print" ($m.print | default "") "materials" (cond (ne ($m.material | default "") "") (slice $m.material) slice) "badges" ($m.badges | default slice) "shops" ($m.shopTypes | default slice) "colours" $cols "note" ($m.description | default "") "priceNotes" ($m.priceNotes | default "") "unit" $unit "qtyUnit" $qtyUnit "thickness" $thick "thicknessTokens"
